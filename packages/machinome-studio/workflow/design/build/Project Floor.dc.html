import cadquery as cq
from solid_node.node import CadQueryNode

from ..parameters import CLEARANCE, MODULE, PLANET_TEETH


class PlanetGear(CadQueryNode):
    def __init__(self, teeth=PLANET_TEETH, module=MODULE, bore=6.0, name=None):
        self.teeth = teeth
        self.module = module
        self.bore = bore + 2 * CLEARANCE
        super().__init__(teeth=teeth, module=module, bore=bore, name=name)

    def render(self):
        # blank first, teeth located from the blank surface
        blank = cq.Workplane("XY").circle(self.pitch_radius).extrude(self.width)
        return blank.union(self.tooth_ring()).faces(">Z").hole(self.bore)
